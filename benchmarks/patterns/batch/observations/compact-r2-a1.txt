import json

def nonnegative(value):
    return type(value) is int and value >= 0

def reject_constant(value):
    raise ValueError(value)

events = {}
with open("events.jsonl", encoding="utf-8") as f:
    for line in f:
        try:
            event = json.loads(line, parse_constant=reject_constant)
        except ValueError:
            continue
        if not isinstance(event, dict):
            continue
        run, seq, kind = (event.get(key) for key in ("run", "seq", "type"))
        if isinstance(run, str) and run and nonnegative(seq) and kind in (
            "item.completed", "turn.completed", "turn.failed"
        ):
            events.setdefault(run, {}).setdefault(seq, event)

runs = []
counters = ("commands", "input_tokens", "output_tokens")
totals = dict.fromkeys(("ok", "failed", "open") + counters, 0)
for run, by_seq in sorted(events.items()):
    report = dict(run=run, status="open", last_message=None)
    report.update(dict.fromkeys(counters, 0))
    for seq in sorted(by_seq):
        event = by_seq[seq]
        kind = event["type"]
        if kind == "item.completed":
            item = event.get("item")
            if isinstance(item, dict):
                if item.get("type") == "command_execution":
                    report["commands"] += 1
                elif item.get("type") == "agent_message" and isinstance(item.get("text"), str):
                    report["last_message"] = item["text"]
            continue
        report["status"] = "failed" if kind == "turn.failed" else "ok"
        usage = event.get("usage")
        if kind == "turn.completed" and isinstance(usage, dict):
            for key in counters[1:]:
                value = usage.get(key)
                if nonnegative(value):
                    report[key] = value
        break
    runs.append(report)
    totals[report["status"]] += 1
    for key in counters:
        totals[key] += report[key]
wj(dict(runs=runs, totals=totals))

def entries(cell):

    if isinstance(value, list):
        yield from (entry for entry in value if isinstance(entry, dict))

from collections import defaultdict
from math import sqrt

train, test = [], []
for row in c("samples.csv"):
    sample = (row["fold"], int(row["x"]), int(row["y"]), row["label"])
    (test if sample[0] == "test" else train).append(sample)

def neighbors(sample, pool, metric):
    _, sx, sy, _ = sample
    distances = []
    for _, px, py, label in pool:
        dx, dy = sx - px, sy - py
        distance = sqrt(dx * dx + dy * dy) if metric == "euclidean" else abs(dx) + abs(dy)
        distances.append((distance, label))
    return t(distances, 5, lambda pair: pair[0])

def predict(near, n, weights):
    scores = defaultdict(int)
    zeros_only = weights == "distance" and near[0][0] == 0
    for distance, label in near[:n]:
        if zeros_only and distance != 0:
            break
        scores[label] += 1 if weights == "uniform" or zeros_only else 1 / distance
    return k(scores, lambda label: (-scores[label], label))

metrics = ("euclidean", "manhattan")
validation = {
    metric: [
        neighbors(sample, (p for p in train if p[0] != sample[0]), metric)
        for sample in train
    ]
    for metric in metrics
}
grid = []
for n in (1, 3, 5):
    for metric in metrics:
        for weights in ("uniform", "distance"):
            correct = sum(
                predict(near, n, weights) == sample[3]
                for near, sample in zip(validation[metric], train)
            )
            grid.append(dict(k=n, metric=metric, weights=weights, correct=correct))

best = x(range(len(grid)), lambda i: grid[i]["correct"])
config = grid[best]
predictions = [
    predict(neighbors(sample, train, config["metric"]), config["k"], config["weights"])
    for sample in test
]
test_correct = sum(label == sample[3] for label, sample in zip(predictions, test))
wj(dict(grid=grid, best=best, predictions=predictions, test_correct=test_correct))
