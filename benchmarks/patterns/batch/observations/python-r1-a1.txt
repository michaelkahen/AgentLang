import ast
import csv
import json
from collections import Counter, defaultdict

def rows(path):
    with open(path, encoding='utf-8', newline='') as f:
        yield from csv.DictReader(f)

def entries(cell):
    if not cell:
        return ()
    try:
        value = json.loads(cell)
    except (ValueError, RecursionError):
        try:
            value = ast.literal_eval(cell)
        except (ValueError, SyntaxError, RecursionError):
            return ()
    return (d for d in value if isinstance(d, dict)) if isinstance(value, list) else ()

films = {
    row['id'] for row in rows('movies.csv') if row['id']
    and any(d.get('name') in ('Science Fiction', 'Sci-Fi')
            for d in entries(row['genres']))
}
credited = set()
credits = Counter()
movies = defaultdict(set)
for row in rows('credits.csv'):
    film = row['id']
    if film in films:
        for crew in entries(row['crew']):
            person = crew.get('id')
            if type(person) is int and person > 0:
                credited.add(film)
                credits[person] += 1
                movies[person].add(film)

leaders = sorted(credits, key=lambda p: (-credits[p], -len(movies[p]), p))[:5]
print(json.dumps(dict(
    films=len(films), credited_films=len(credited), appearances=sum(credits.values()),
    leaders=[dict(id=p, credits=credits[p], movies=len(movies[p])) for p in leaders],
)))

import csv
import json
from collections import defaultdict
from itertools import product
from math import sqrt

with open('samples.csv', encoding='utf-8', newline='') as f:
    rows = [(r['fold'], int(r['x']), int(r['y']), r['label'])
            for r in csv.DictReader(f)]
train = [r for r in rows if r[0] != 'test']
test = [r for r in rows if r[0] == 'test']

def neighbors(row, pool, metric):
    _, x, y, _ = row
    distances = [
        (sqrt((x-a)**2 + (y-b)**2) if metric == 'euclidean'
         else abs(x-a) + abs(y-b), label)
        for _, a, b, label in pool
    ]
    return sorted(distances, key=lambda p: p[0])[:5]

def vote(near, k, weights):
    scores = defaultdict(float)
    zero_only = weights == 'distance' and near[0][0] == 0
    for d, label in near[:k]:
        if zero_only and d != 0:
            break
        scores[label] += 1 if zero_only or weights == 'uniform' else 1/d
    return min(scores, key=lambda label: (-scores[label], label))

metrics = ['euclidean', 'manhattan']
cv = {m: [neighbors(r, (s for s in train if s[0] != r[0]), m)
          for r in train] for m in metrics}
grid = [
    dict(k=k, metric=m, weights=w,
         correct=sum(vote(n, k, w) == r[3] for n, r in zip(cv[m], train)))
    for k, m, w in product([1, 3, 5], metrics, ['uniform', 'distance'])
]
best = max(range(len(grid)), key=lambda i: grid[i]['correct'])
config = grid[best]
predictions = [vote(neighbors(r, train, config['metric']), config['k'],
                    config['weights']) for r in test]
print(json.dumps(dict(grid=grid, best=best, predictions=predictions,
                      test_correct=sum(p == r[3] for p, r in zip(predictions, test)))))

import csv
import hashlib
import io
import json
from itertools import groupby
from pathlib import Path

factor = int(input())
report = {}

def encode(value):
    return (json.dumps(value) + '\n').encode('utf-8')

def record(task, count, data, golden):
    report[task] = dict(records=count * factor, bytes=len(data),
                        input_sha256=hashlib.sha256(data).hexdigest(),
                        golden_sha256=hashlib.sha256(golden).hexdigest())

for task, source in [('late_flights', 'data/tweets.csv'),
                     ('goal_gap', 'data/projects.csv')]:
    with open(source, encoding='utf-8', newline='') as f:
        header, *rows = csv.reader(f)
    out = io.StringIO(newline='')
    writer = csv.writer(out, lineterminator='\n')
    writer.writerow(header)
    writer.writerows(rows * factor)
    golden = Path(task + '.out').read_bytes()
    if task == 'late_flights':
        golden = f'{int(golden) * factor}\n'.encode('utf-8')
    record(task, len(rows), out.getvalue().encode('utf-8'), golden)

for task in ['word_counts', 'sort_scores']:
    rows = json.loads(Path(task + '.in').read_text(encoding='utf-8'))
    golden = Path(task + '.out').read_bytes()
    if task == 'sort_scores':
        scaled = []
        for _, group in groupby(json.loads(golden), key=lambda p: p[1]):
            scaled.extend(list(group) * factor)
        golden = encode(scaled)
    record(task, len(rows), encode(rows * factor), golden)

print(json.dumps(report))

import json
from collections import defaultdict

def natural(value):
    return type(value) is int and value >= 0

events = defaultdict(dict)
with open('events.jsonl', encoding='utf-8') as f:
    for line in f:
        try:
            event = json.loads(line)
        except (ValueError, RecursionError):
            continue
        if not isinstance(event, dict):
            continue
        run, seq, kind = (event.get(key) for key in ('run', 'seq', 'type'))
        if (isinstance(run, str) and run and natural(seq)
                and kind in ('item.completed', 'turn.completed', 'turn.failed')):
            events[run].setdefault(seq, event)

runs = []
numeric = ('commands', 'input_tokens', 'output_tokens')
totals = dict.fromkeys(('ok', 'failed', 'open', *numeric), 0)
for run, kept in sorted(events.items()):
    result = dict(run=run, status='open', last_message=None, commands=0,
                  input_tokens=0, output_tokens=0)
    for seq in sorted(kept):
        event = kept[seq]
        kind = event['type']
        if kind == 'turn.failed':
            result['status'] = 'failed'
            break
        if kind == 'turn.completed':
            result['status'] = 'ok'
            usage = event.get('usage')
            if isinstance(usage, dict):
                for key in numeric[1:]:
                    value = usage.get(key)
                    if natural(value):
                        result[key] = value
            break
        item = event.get('item')
        if isinstance(item, dict):
            if item.get('type') == 'command_execution':
                result['commands'] += 1
            elif item.get('type') == 'agent_message' and isinstance(item.get('text'), str):
                result['last_message'] = item['text']
    runs.append(result)
    totals[result['status']] += 1
    for key in numeric:
        totals[key] += result[key]

print(json.dumps(dict(runs=runs, totals=totals)))
