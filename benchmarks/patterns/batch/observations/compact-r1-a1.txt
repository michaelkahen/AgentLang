from collections import defaultdict
from math import sqrt

train, test = [], []
for i, row in enumerate(c("samples.csv")):
    fold = row["fold"]
    (test if fold == "test" else train).append(
        (i, fold if fold == "test" else int(fold), int(row["x"]), int(row["y"]), row["label"]))

def neighbors(row, pool, metric):
    _, _, a, b, _ = row
    result = []
    for i, _, x, y, label in pool:
        dx, dy = abs(x-a), abs(y-b)
        dist = sqrt(dx*dx + dy*dy) if metric == "euclidean" else dx+dy
        result.append((dist, i, label))
    return sorted(result)[:5]

def vote(near, k, weights):
    scores = defaultdict(int)
    zero_only = weights == "distance" and near[0][0] == 0
    for dist, _, label in near[:k]:
        if not zero_only or dist == 0:
            scores[label] += 1 if weights == "uniform" or zero_only else 1/dist
    return min(scores, key=lambda label: (-scores[label], label))

metrics = ("euclidean", "manhattan")
nearby = {m: [neighbors(row, (p for p in train if p[1] != row[1]), m)
              for row in train] for m in metrics}
grid = [dict(k=k, metric=m, weights=w,
             correct=sum(vote(near, k, w) == row[4] for near, row in zip(nearby[m], train)))
        for k in (1, 3, 5) for m in metrics for w in ("uniform", "distance")]
best = max(range(len(grid)), key=lambda i: grid[i]["correct"])
config = grid[best]
predictions = [vote(neighbors(row, train, config["metric"]), config["k"], config["weights"])
               for row in test]
wj(dict(grid=grid, best=best, predictions=predictions,
        test_correct=sum(label == row[4] for label, row in zip(predictions, test))))

import json
from collections import defaultdict

def nonnegative(value):
    return type(value) is int and value >= 0

def reject_constant(value):
    raise ValueError(value)

events = defaultdict(dict)
with open("events.jsonl", encoding="utf-8") as f:
    for line in f:
        try:
            event = json.loads(line, parse_constant=reject_constant)
        except (ValueError, RecursionError):
            continue
        if not isinstance(event, dict):
            continue
        run, seq, kind = (event.get(key) for key in ("run", "seq", "type"))
        if isinstance(run, str) and run and nonnegative(seq) and kind in (
            "item.completed", "turn.completed", "turn.failed"
        ):
            events[run].setdefault(seq, event)

runs = []
tokens = ("input_tokens", "output_tokens")
totals = dict(ok=0, failed=0, open=0, commands=0, input_tokens=0, output_tokens=0)
for run in sorted(events):
    row = dict(run=run, status="open", last_message=None, commands=0,
               input_tokens=0, output_tokens=0)
    for _, event in sorted(events[run].items()):
        kind = event["type"]
        if kind != "item.completed":
            row["status"] = "ok" if kind == "turn.completed" else "failed"
            usage = event.get("usage")
            if kind == "turn.completed" and isinstance(usage, dict):
                for key in tokens:
                    value = usage.get(key)
                    if nonnegative(value):
                        row[key] = value
            break
        item = event.get("item")
        if isinstance(item, dict):
            if item.get("type") == "agent_message" and isinstance(item.get("text"), str):
                row["last_message"] = item["text"]
            elif item.get("type") == "command_execution":
                row["commands"] += 1
    runs.append(row)
    totals[row["status"]] += 1
    for key in ("commands", *tokens):
        totals[key] += row[key]
wj(dict(runs=runs, totals=totals))
